import { defineComponent, h as create, onMounted, onUpdated, reactive } from "vue";
import * as K from "@kjun-ui/vue3";
import "@kjun-ui/vue3/styles.css";
import "./kjun.css";
/* MODULE */
// Recipes use React-style props; Vue 3 takes v-model props, update events and slots.
const unitless = new Set(["lineHeight", "flex", "flexGrow", "flexShrink", "opacity", "zIndex", "fontWeight", "order", "aspectRatio"]);
const css = style => style && Object.fromEntries(Object.entries(style).map(([key, value]) => [key, typeof value === "number" && !unitless.has(key) ? value + "px" : value]));
const list = children => Array.isArray(children) ? children : children == null ? [] : [children];
const slotNames = { errorContent: "error", emptyContent: "empty", loadingContent: "loading", "selection-toolbar": "selectionToolbar" };
const renderSlots = { renderCell: fn => ({ value, column, row, index }) => fn(value, column, row, index), renderExpand: fn => ({ row, index }) => fn(row, index), renderSelectionToolbar: fn => ({ selected }) => fn(selected) };
const renderSlotNames = { renderCell: "cell", renderExpand: "expand", renderSelectionToolbar: "selectionToolbar" };
function vueProps(name, props) {
  const component = K[name], declared = component?.props || {}, mapped = {}, slots = {};
  for (const [key, value] of Object.entries(props)) {
    if (key in renderSlots) { if (value) slots[renderSlotNames[key]] = renderSlots[key](value); continue; }
    if (key === "value" && "modelValue" in declared) mapped.modelValue = value;
    else if (key === "onValueChange") mapped["onUpdate:" + ("modelValue" in declared ? "modelValue" : "value")] = value;
    else if (key === "onOpenChange") mapped["onUpdate:open"] = value;
    else if (key === "onChangeCommit") mapped.onChange = value;
    else if (key === "onSelectionChange") mapped["onUpdate:selected"] = value;
    else if (key === "onExpandedRowsChange") mapped["onUpdate:expandedRows"] = value;
    else if (key === "onSortChange") mapped["onUpdate:sort"] = value;
    else if (key === "onPageChange" && name === "KjunPagination") mapped["onUpdate:page"] = value;
    else if (key === "onPageSizeChange") mapped["onUpdate:pageSize"] = value;
    else if (key === "style") mapped.style = css(value);
    else mapped[key] = value;
  }
  return { mapped, slots };
}
const Example = defineComponent({
  /* VUE_PROPS */
  setup(exampleProps) {
    const values = reactive(__KJUN_VALUES__);
    const set = (key, value) => { values[key] = value; };
    const feedback = K.useKjunFeedback();
    /* VUE_OBSERVE */
    return () => {
      // Recipes read values with hasOwnProperty, which Vue does not track; reading the keys renders again when one is added.
      void Object.keys(values);
      const settings = __KJUN_SETTINGS__;
      const rawH = (name, props = {}, children, slots = {}) => {
        if (name === "ExampleFailure") return create(defineComponent(() => () => {
          if (props.failed) throw new Error("예제에서 의도적으로 발생시킨 렌더링 오류입니다.");
          return create(K.KjunAlert, { type: "success" }, { default: () => "정상 콘텐츠를 표시합니다." });
        }));
        if (name === "ScrollItems") return create("div", { style: { display: "flex", flexWrap: "nowrap", gap: "12px", width: "max-content" } }, list(children));
        if (name === "Status") return create("p", { ref: props.ref, role: "status", "aria-live": "polite", "aria-atomic": "true", tabindex: -1, style: { margin: 0 } }, list(children));
        // Documentation-only layout: package components keep their own geometry.
        if (name === "CaseScreen") return create("div", { "data-testid": "design-screen", style: { height: "520px", position: "relative", display: "flex", flexDirection: "column", overflow: "hidden", border: "1px solid var(--kjun-divider)" } }, list(children));
        if (name === "CaseViewport") return create("div", { style: { position: "relative", display: "flex", flexDirection: "column", flex: 1, minHeight: 0, marginBottom: (props.inset || 0) + "px" } }, list(children));
        if (name === "CaseScroll") return create("div", { "data-testid": "design-scroll", ref: node => { if (node && !node.dataset.positioned) { node.scrollTop = node.scrollHeight; node.dataset.positioned = "true"; } }, style: { flex: 1, minHeight: 0, overflow: "auto", padding: "16px" } }, list(children));
        if (name === "CaseFooter") return create("div", { "data-testid": "design-footer", style: { background: "var(--kjun-surface)", flexShrink: 0, ...(props.overlay ? { position: "absolute", left: 0, right: 0, bottom: 0 } : {}) } }, list(children));
        if (name === "CaseKeyboard") return create("div", { "data-testid": "design-keyboard", style: { position: "absolute", left: 0, right: 0, bottom: 0, height: "220px", display: "flex", alignItems: "center", justifyContent: "center", background: "var(--kjun-secondary)", borderTop: "1px solid var(--kjun-divider)", zIndex: 1 } }, list(children));
        if (name === "CaseColumns") return create("div", { style: { display: "grid", gridTemplateColumns: `repeat(${props.columns}, minmax(0, 1fr))`, gap: "16px" } }, list(children));
        if (name === "CaseCell" || name === "CasePart") return create("div", { "data-testid": props.id ? "design-" + props.id : undefined, style: { minWidth: 0, ...(props.inline ? { display: "inline-flex" } : {}), ...(props.scroll ? { overflowX: "auto" } : {}) } }, list(children));
        if (name === "FieldStack") return create("div", { "data-kjun-layout": "form", style: { display: "flex", flexDirection: "column", gap: "20px" } }, list(children));
        // Scroll bodies keep lists full-bleed; Inset aligns other content with list text.
        if (name === "Inset") return create("div", { style: { paddingInline: "16px" } }, list(children));
        if (["Screen", "Scroll", "Box"].includes(name)) return create("div", { style: name === "Screen" ? { height: (props.height || 520) + "px", display: "flex", flexDirection: "column", overflow: "hidden", border: "1px solid var(--kjun-divider)" } : name === "Scroll" ? { flex: 1, minHeight: 0, overflow: "auto", padding: "16px 0", display: "flex", flexDirection: "column", gap: "16px" } : { width: props.width + "px", maxWidth: "100%", ...(props.height ? { height: props.height + "px", position: "relative", transform: "translateZ(0)" } : {}) } }, list(children));
        if (["Group", "Stack", "Text"].includes(name)) return create(name === "Text" ? "span" : "div", { ...props, class: "catalog-" + name.toLowerCase(), style: css(props.style) }, list(children));
        const { mapped, slots: rendered } = vueProps(name, props);
        const named = Object.fromEntries(Object.entries(slots).filter(([, value]) => value != null)
          .map(([key, value]) => [slotNames[key] || key, typeof value === "function" ? value : () => value]));
        const body = list(children);
        return create(K[name], mapped, { ...(body.length ? { default: () => body } : {}), ...rendered, ...named });
      };
      const visual = JSON.parse(settings.visual || "{}");
      const visualH = (name, props = {}, children, slots = {}) => {
        const spec = visual[name];
        return rawH(name, { ...props, ...spec?.props }, spec?.text ?? children, slots);
      };
      const h = visualH;
      return create("section", { class: "catalog-example", "data-component": "__KJUN_NAME__" }, [
        create("div", { class: "catalog-render" }, [renderExample(h, values, set, feedback, settings)]),
        create("output", { role: "status" }, values.message || ""),
      ]);
    };
  },
});
export default defineComponent({
  /* VUE_PROPS */
  setup(exampleProps) {
    return () => create(K.KjunProvider, null, { default: () => create(K.KjunFeedbackProvider, null, { default: () => create(Example, /* VUE_EXAMPLE_PROPS */) }) });
  },
});
